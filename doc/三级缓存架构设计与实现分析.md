# 三级缓存架构设计与实现分析报告

## 一、概述

本文档详细分析 WanAndroid 项目中网络请求三级缓存的技术实现，包括架构设计、代码结构和工作流程。

### 1.1 什么是三级缓存

三级缓存是指按层级划分的缓存体系：

```
┌──────────────────────────────────────────────┐
│           第三级: Network (网络)               │
│     最后一步，网络请求获取原始数据              │
└──────────────────────────────────────────────┘
                      ▲
                      │
┌──────────────────────────────────────────────┐
│           第二级: Disk (磁盘)                │
│     持久化缓存，应用重启后依然有效           │
└──────────────────────────────────────────────┘
                      ▲
                      │
┌──────────────────────────────────────────────┐
│           第一级: Memory (内存)              │
│     进程级缓存，最快，退出应用后失效         │
└──────────────────────────────────────────────┘
```

### 1.2 缓存策略

**核心策略：有缓存立即返回，同时异步更新网络数据**

```
用户请求 → [有缓存?] ──是──→ 立即返回缓存 → 异步请求网络 → 更新缓存
          │
          └──否──→ 请求网络 → 返回数据 → 写入缓存
```

## 三、核心实现

### 3.1 缓存接口

```kotlin
// RequestCache.kt
interface RequestCache {
    suspend fun get(cacheKey: String): String?
    suspend fun put(cacheKey: String, data: String, expireTime: Long = 30 * 24 * 60 * 60 * 1000L)
    suspend fun clear(cacheKey: String? = null)
    suspend fun clearExpired()
}
```

### 3.2 缓存数据模型

```kotlin
// CacheItem.kt
data class CacheItem(
    val data: String,          // 缓存的数据 (JSON)
    val timestamp: Long,    // 缓存时间戳
    val expireTime: Long     // 过期时间 (毫秒)
) {
    val isExpired: Boolean
        get() = System.currentTimeMillis() - timestamp > expireTime
}
```

### 3.3 一级缓存 (内存)

```kotlin
// MemoryCache.kt
class MemoryCache : RequestCache {
    private val cache = mutableMapOf<String, CacheItem>()
    private val lock = Mutex()

    override suspend fun get(cacheKey: String): String? = lock.withLock {
        val item = cache[cacheKey] ?: return@withLock null
        if (item.isExpired) {
            cache.remove(cacheKey)
            return@withLock null
        }
        item.data
    }

    override suspend fun put(cacheKey: String, data: String, expireTime: Long) = lock.withLock {
        cache[cacheKey] = CacheItem(data, System.currentTimeMillis(), expireTime)
    }
    // ...
}
```

**特点**：
- 使用 `MutableMap` 存储
- 进程内有效，退出应用后失效
- 线程安全，使用 `Mutex`

### 3.4 二级缓存 (磁盘)

```kotlin
// DiskCache.kt
object DiskCache : RequestCache {
    private var cacheDir: File? = null
    private val lock = Mutex()

    fun init(ctx: Context) {
        cacheDir = File(ctx.cacheDir, "network_cache").apply { mkdirs() }
    }

    override suspend fun get(cacheKey: String): String? = lock.withLock {
        withContext(Dispatchers.IO) {
            val file = File(cacheDir, cacheKey)
            if (!file.exists()) return@withContext null

            val cacheItem = try {
                file.readText().let { gson.fromJson(it, CacheItem::class.java) }
            } catch (e: Exception) { null }

            if (cacheItem?.isExpired == true) {
                file.delete()
                return@withContext null
            }
            cacheItem?.data
        }
    }
    // ...
}
```

**特点**：
- 单例模式，使用文件存储
- 缓存目录：`/data/data/{package}/cache/network_cache`
- 线程安全，使用 `Mutex` + `Dispatchers.IO`

### 3.5 混合缓存 (L1 + L2)

```kotlin
// CompositeCache.kt
class CompositeCache(
    private val memoryCache: MemoryCache,
    private val diskCache: RequestCache
) : RequestCache {

    override suspend fun get(cacheKey: String): String? {
        // 1. 先从内存获取
        val memoryData = memoryCache.get(cacheKey)
        if (memoryData != null) return memoryData

        // 2. 内存没有再从磁盘获取
        val diskData = diskCache.get(cacheKey)
        if (diskData != null) {
            // 3. 磁盘命中，同步到内存
            memoryCache.put(cacheKey, diskData, 0)
            return diskData
        }
        return null
    }

    override suspend fun put(cacheKey: String, data: String, expireTime: Long) {
        // 同时写入两层
        memoryCache.put(cacheKey, data, expireTime)
        diskCache.put(cacheKey, data, expireTime)
    }
}
```

**策略**：
- 获取：Memory → Disk
- 存储：Memory && Disk

### 4.2 缓存策略 - 核心方法

```kotlin
private suspend inline fun <T> getWithCache(
    cacheKey: String,
    expireTime: Long,
    crossinline fetch: suspend () -> T
): T where T : Any {
    // 1. 缓存禁用时直接请求网络
    if (!CACHE_ENABLE) return fetch()

    // 2. 先从缓存获取
    val cachedData = cache.get(cacheKey)
    if (cachedData != null) {
        val cached = try {
            gson.fromJson(cachedData, CacheResult::class.java)
        } catch (e: Exception) { null }

        if (cached != null) {
            // 3. 有缓存，立即返回
            // 4. 同时异步请求网络更新缓存
            repositoryScope.launch {
                try {
                    val freshData = fetch()
                    val json = gson.toJson(CacheResult(data = freshData))
                    cache.put(cacheKey, json, expireTime)
                } catch (e: Exception) {
                    LogUtils.eTag(TAG, "refresh cache error: ${e.message}")
                }
            }
            return cached.data as T
        }
    }

    // 5. 无缓存，请求网络并缓存
    val data = fetch()
    if (data != null) {
        val json = gson.toJson(CacheResult(data = data))
        cache.put(cacheKey, json, expireTime)
    }
    return data
}
```

### 4.3 缓存接口配置

```kotlin
private const val CACHE_TIME_BANNER = 30 * 60 * 1000L      // 30分钟
private const val CACHE_TIME_TOP_LIST = 30 * 60 * 1000L   // 30分钟
private const val CACHE_TIME_TREE = 24 * 60 * 60 * 1000L   // 24小时
private const val CACHE_TIME_LIST = 10 * 60 * 1000L    // 10分钟

// Banner 使用缓存
suspend fun getBanner() = withContext(Dispatchers.IO) {
    getWithCache("banner", CACHE_TIME_BANNER) {
        apiService.getBanner().apiData()!!
    }
}

// 项目分类树使用缓存
suspend fun getProjectTree() = withContext(Dispatchers.IO) {
    getWithCache("project_tree", CACHE_TIME_TREE) {
        apiService.getProjectTree().apiData()!!
    }
}

// 项目列表使用缓存
suspend fun getProjectList(id: Int, page: Int) = withContext(Dispatchers.IO) {
    val cacheKey = "project_list_${id}_$page"
    getWithCache(cacheKey, CACHE_TIME_LIST) {
        apiService.getProjectList(id = id, page = page).apiData()!!
    }
}
```

## 七、线程安全设计

### 7.1 Mutex 锁

所有缓存实现都使用 `kotlinx.coroutines.sync.Mutex` 保证线程安全：

```kotlin
private val lock = Mutex()

override suspend fun get(cacheKey: String): String? {
    return lock.withLock {
        // 临界区代码
    }
}
```

### 7.2 Dispatchers.IO

磁盘操作在 IO 线程池执行，避免阻塞主线程：

```kotlin
withContext(Dispatchers.IO) {
    // 磁盘读写操作
}
```

## 八、性能优化

### 8.1 缓存优势

1. **减少网络请求**：有缓存时直接返回，无需等待网络
2. **提升用户体验**：首次加载立即响应
3. **节省流量**：异步更新，不需要用户手动刷新

### 8.2 缓存时间配置原则

- **高频变化数据**：30分钟 (Banner, TopList)
- **低频变化数据**：24小时 (分类树)
- **中等频率数据**：10分钟 (列表)
